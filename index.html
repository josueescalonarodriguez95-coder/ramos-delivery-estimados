Zona designada: ${datos.destinoDel} ${formatCurrency(currentDeliveryTotal)} `; } if (currentTab === 'install' || currentTab === 'both') { tableContent += ` Servicio de Instalación de Arte
Ubicación: ${datos.zonaInst}
Detalle de obras: ${datos.obrasInfo} ${formatCurrency(currentInstallTotal)} `; } let discountHtml = ''; if (discountApplied) { discountHtml = `
Descuento Promocional (10%): -${formatCurrency(globalSubtotal * 0.10)}
`; } let serviceTypeLabel = currentTab === 'delivery' ? 'Solo Delivery' : (currentTab === 'install' ? 'Instalación de Arte' : 'Servicio Completo (Delivery + Instalación)'); const pdfTemplateStr = `

Ramos Delivery Enterprise

Miami, Florida

Tel: (786) 426-2444

INVOICE

Factura #: ${invNum}

Fecha: ${fechaString}

Facturar a:

Cliente: ${datos.clientName}

Detalles del Servicio:

Especialista: ${datos.workerNameFormat}

Tipo de Servicio: ${serviceTypeLabel}

${tableContent}
Descripción del Concepto Monto
Subtotal Estimado: ${formatCurrency(globalSubtotal)}
${discountHtml}
Total General: ${formatCurrency(globalTotal)}

Gracias por confiar en Ramos Delivery Enterprise para el manejo logístico de su arte.

Para consultas sobre esta factura, por favor comuníquese directamente con su especialista o contáctenos al (786) 426-2444.

`; // --- AQUÍ ESTÁ LA MAGIA --- const element = document.createElement('div'); element.innerHTML = pdfTemplateStr; // Lo posicionamos fuera de la pantalla para que el usuario no lo vea element.style.position = 'absolute'; element.style.left = '-9999px'; element.style.top = '-9999px'; element.style.zIndex = '-1'; // LO AÑADIMOS AL DOM. Sin esto, el PDF sale en blanco. document.body.appendChild(element); // --------------------------- const opt = { margin: 0.1, filename: `Invoice_${datos.clientName.replace(/\s+/g, '_')}_${invNum}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: true, scrollY: 0, windowWidth: 800 }, jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' } }; html2pdf().set(opt).from(element).outputPdf('blob').then(function(pdfBlob) { // --- LIMPIEZA --- // Borramos el div temporal para no ensuciar la página document.body.removeChild(element); // ---------------- const file = new File([pdfBlob], opt.filename, { type: 'application/pdf' }); if (navigator.share && navigator.canShare && navigator.canShare({ files: [file] })) { navigator.share({ files: [file], title: 'Invoice Ramos Delivery', text: 'Adjunto el invoice de su servicio con Ramos Delivery Enterprise.' }).then(() => { btnPdf.innerHTML = originalContent; }).catch((error) => { console.log('Compartir cancelado o con error', error); btnPdf.innerHTML = originalContent; }); } else { const url = URL.createObjectURL(pdfBlob); const a = document.createElement('a'); a.href = url; a.download = opt.filename; document.body.appendChild(a); a.click(); URL.revokeObjectURL(url); btnPdf.innerHTML = originalContent; } }).catch(err => { // En caso de error, también debemos limpiar el DOM document.body.removeChild(element); btnPdf.innerHTML = originalContent; console.error('Error al generar PDF:', err); }); }
